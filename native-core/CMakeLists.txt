option(SYSPULSE_NATIVE_X11 "Build the experimental native X11 dashboard" ON)
if(SYSPULSE_NATIVE_X11 AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(X11 QUIET)
  if(X11_FOUND)
    add_executable(syspulse-dashboard-x11 src/dashboard_x11.cpp)
    target_include_directories(syspulse-dashboard-x11 PRIVATE ${X11_INCLUDE_DIR})
    target_link_libraries(syspulse-dashboard-x11 PRIVATE syspulse_native_core ${X11_LIBRARIES})
    if(BUILD_TESTING)
      find_program(XVFB_RUN xvfb-run)
      if(XVFB_RUN)
        add_test(NAME native-dashboard-x11 COMMAND ${XVFB_RUN} -a $<TARGET_FILE:syspulse-dashboard-x11> --smoke)
        set_tests_properties(native-dashboard-x11 PROPERTIES TIMEOUT 15)
      endif()
    endif()
  endif()
endif()
