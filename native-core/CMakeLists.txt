cmake_minimum_required(VERSION 3.16)
project(SysPulseNativeCore LANGUAGES CXX)
add_library(syspulse_native_core STATIC src/metrics.cpp)
target_include_directories(syspulse_native_core PUBLIC include)
target_compile_features(syspulse_native_core PUBLIC cxx_std_17)
set_target_properties(syspulse_native_core PROPERTIES CXX_EXTENSIONS OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(native_metrics_test tests/test_metrics.cpp)
  target_link_libraries(native_metrics_test PRIVATE syspulse_native_core)
  add_test(NAME native-metrics COMMAND native_metrics_test)
endif()
