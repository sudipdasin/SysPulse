cmake_minimum_required(VERSION 3.16)
project(SysPulseNativeCore LANGUAGES CXX)
add_library(syspulse_native_core STATIC src/metrics.cpp src/inspection.cpp)
target_include_directories(syspulse_native_core PUBLIC include)
target_compile_features(syspulse_native_core PUBLIC cxx_std_17)
set_target_properties(syspulse_native_core PROPERTIES CXX_EXTENSIONS OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(native_metrics_test tests/test_metrics.cpp)
  target_link_libraries(native_metrics_test PRIVATE syspulse_native_core)
  add_test(NAME native-metrics COMMAND native_metrics_test)
endif()
add_executable(syspulse-inspect src/inspect_main.cpp)
target_link_libraries(syspulse-inspect PRIVATE syspulse_native_core)
if(BUILD_TESTING)
  add_executable(native_inspection_test tests/test_inspection.cpp)
  target_link_libraries(native_inspection_test PRIVATE syspulse_native_core)
  add_test(NAME native-inspection COMMAND native_inspection_test)
endif()
target_sources(syspulse_native_core PRIVATE src/runtime.cpp)
if(BUILD_TESTING)
  add_executable(native_runtime_test tests/test_runtime.cpp)
  target_link_libraries(native_runtime_test PRIVATE syspulse_native_core)
  add_test(NAME native-runtime COMMAND native_runtime_test)
endif()
target_sources(syspulse_native_core PRIVATE src/device_rates.cpp)
if(BUILD_TESTING)
  add_executable(native_device_rates_test tests/test_device_rates.cpp)
  target_link_libraries(native_device_rates_test PRIVATE syspulse_native_core)
  add_test(NAME native-device-rates COMMAND native_device_rates_test)
endif()
add_executable(syspulse-sample src/sample_main.cpp)
target_link_libraries(syspulse-sample PRIVATE syspulse_native_core)
target_sources(syspulse_native_core PRIVATE src/health.cpp)
if(BUILD_TESTING)
  add_executable(native_health_test tests/test_health.cpp)
  target_link_libraries(native_health_test PRIVATE syspulse_native_core)
  add_test(NAME native-health COMMAND native_health_test)
endif()
option(SYSPULSE_NATIVE_X11 "Build the experimental native X11 dashboard" ON)
if(SYSPULSE_NATIVE_X11 AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(X11 QUIET)
  if(X11_FOUND)
    add_executable(syspulse-dashboard-x11 src/dashboard_x11.cpp)
    target_include_directories(syspulse-dashboard-x11 PRIVATE ${X11_INCLUDE_DIR})
    target_link_libraries(syspulse-dashboard-x11 PRIVATE syspulse_native_core ${X11_LIBRARIES})
    if(BUILD_TESTING)
      find_program(XVFB_RUN xvfb-run)
      if(XVFB_RUN)
        add_test(NAME native-dashboard-x11 COMMAND ${XVFB_RUN} -a $<TARGET_FILE:syspulse-dashboard-x11> --smoke)
        set_tests_properties(native-dashboard-x11 PROPERTIES TIMEOUT 15)
      endif()
    endif()
  endif()
endif()
