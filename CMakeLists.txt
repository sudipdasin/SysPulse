# Only public, operator-verified trust anchors are packaged. Evaluation builds omit them by default.
foreach(channel alpha beta stable)
    string(TOUPPER "${channel}" upper)
    set(SYSPULSE_${upper}_PUBLIC_KEY "" CACHE FILEPATH "Binary OpenPGP public release key for ${channel}")
    set(SYSPULSE_${upper}_KEY_SHA256 "" CACHE STRING "Independently verified public key SHA256 for ${channel}")
endforeach()
option(SYSPULSE_USE_OWNER_ALPHA_TRUST "Include the verified owner-provided Alpha release public key" OFF)
if(SYSPULSE_USE_OWNER_ALPHA_TRUST)
    if(SYSPULSE_ALPHA_PUBLIC_KEY OR SYSPULSE_ALPHA_KEY_SHA256)
        message(FATAL_ERROR "Owner trust and external Alpha trust options are mutually exclusive")
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(owner_key "${CMAKE_CURRENT_BINARY_DIR}/owner-alpha-public.gpg")
    set(owner_pin "1d16b08d5bf50ba03cd8f294a94ef7f5259f9c5f4795a39134e9fde4b5968dc6")
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/decode-owner-alpha-key.py"
        "${CMAKE_CURRENT_SOURCE_DIR}/assets/update-trust/alpha-public.gpg.b64" "${owner_key}" "${owner_pin}"
        RESULT_VARIABLE owner_result)
    if(NOT owner_result EQUAL 0)
        message(FATAL_ERROR "Owner Alpha trust anchor did not match the approved digest")
    endif()
    set(SYSPULSE_ALPHA_PUBLIC_KEY "${owner_key}")
    set(SYSPULSE_ALPHA_KEY_SHA256 "${owner_pin}")
endif()
foreach(channel alpha beta stable)
    string(TOUPPER "${channel}" upper)
    set(key "${SYSPULSE_${upper}_PUBLIC_KEY}")
    set(pin "${SYSPULSE_${upper}_KEY_SHA256}")
    if(key OR pin)
        if(NOT key OR NOT pin)
            message(FATAL_ERROR "Both public key and independent SHA256 required for ${channel}")
        endif()
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/validate-update-key.py" "${key}" "${pin}"
                        RESULT_VARIABLE trust_result)
        if(NOT trust_result EQUAL 0)
            message(FATAL_ERROR "Invalid public release trust anchor for ${channel}")
        endif()
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${channel}.sha256" "${pin}\n")
        install(FILES "${key}" DESTINATION share/syspulse/update-keys RENAME "${channel}.gpg")
        install(FILES "${CMAKE_CURRENT_BINARY_DIR}/${channel}.sha256" DESTINATION share/syspulse/update-keys)
    endif()
endforeach()
