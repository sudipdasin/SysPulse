cmake_minimum_required(VERSION 3.16)
project(SysPulse VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(syspulse src/main.cpp)
target_link_libraries(syspulse PRIVATE X11::X11)
target_compile_options(syspulse PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME version COMMAND syspulse --version)
add_test(NAME diagnostics COMMAND syspulse --diagnostics)
add_test(NAME updater COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/update-test.sh)
install(TARGETS syspulse DESTINATION bin)
install(PROGRAMS tools/syspulse-update DESTINATION bin)

add_executable(health-test tests/health-test.cpp)
target_include_directories(health-test PRIVATE src)
add_test(NAME memory_parser COMMAND health-test)
