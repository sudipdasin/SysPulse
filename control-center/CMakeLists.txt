cmake_minimum_required(VERSION 3.16)
project(SysPulseControlCenter VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(X11 REQUIRED)
add_executable(syspulse-control-center main.cpp)
target_link_libraries(syspulse-control-center PRIVATE X11::X11)
target_compile_options(syspulse-control-center PRIVATE -Wall -Wextra -Wpedantic)
add_executable(control-center-test test.cpp)
enable_testing()
add_test(NAME control_catalog COMMAND control-center-test)
add_test(NAME control_diagnostics COMMAND syspulse-control-center --diagnose)
add_test(NAME control_version COMMAND syspulse-control-center --version)
install(TARGETS syspulse-control-center DESTINATION bin)
